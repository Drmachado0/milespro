**2.1 — Create `src/components/layout/CrispWidget.tsx`:**

```tsx
import { useEffect } from 'react';
import { Crisp } from 'crisp-sdk-web';
import { useAuth } from '@/hooks/useAuth';
import { useConsent } from '@/hooks/useConsent';
import { logger } from '@/lib/logger';

const WEBSITE_ID = import.meta.env.VITE_CRISP_WEBSITE_ID;

export function CrispWidget(): null {
  const { user } = useAuth();
  const { consent } = useConsent();

  useEffect(() => {
    if (!WEBSITE_ID) {
      logger.warn('[Crisp] VITE_CRISP_WEBSITE_ID unset — widget disabled');
      return;
    }
    // Gate behind privacy_accepted_at (essential cookies for helpdesk are OK,
    // but we still want the user to have seen the privacy notice listing Crisp).
    if (!consent?.privacy_accepted_at) return;

    Crisp.configure(WEBSITE_ID, {
      autoload: true,
      tokenId: user?.id ?? undefined,
    });

    if (user?.email) {
      // Crisp needs email for the inbox to thread messages; this IS sent to Crisp
      // (declared in Privacidade.tsx sub-processors per W1a Task 7).
      Crisp.user.setEmail(user.email);
    }

    return () => {
      try { Crisp.chat.hide(); } catch { /* noop */ }
    };
  }, [user?.id, user?.email, consent?.privacy_accepted_at]);

  return null;
}
```
