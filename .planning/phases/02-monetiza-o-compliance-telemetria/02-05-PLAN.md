-- Phase 2 W2a — CRIT-04 idempotency primitive
--
-- The (provider, event_id) UNIQUE constraint IS the idempotency mechanism.
-- INSERT-first then side-effects pattern in the webhook handler relies on
-- this constraint throwing 23505 (unique_violation) on duplicate delivery.

CREATE TABLE public.webhook_events (
  id              UUID         NOT NULL DEFAULT gen_random_uuid() PRIMARY KEY,
  provider        TEXT         NOT NULL CHECK (provider IN ('asaas')),
  event_id        TEXT         NOT NULL,
  event_type      TEXT         NOT NULL,
  payload         JSONB        NOT NULL,
  status          TEXT         NOT NULL DEFAULT 'received'
                  CHECK (status IN ('received', 'processing', 'processed', 'failed')),
  processed_at    TIMESTAMPTZ,
  error_message   TEXT,
  retries         INTEGER      NOT NULL DEFAULT 0,
  created_at      TIMESTAMPTZ  NOT NULL DEFAULT now(),
  CONSTRAINT webhook_events_unique UNIQUE (provider, event_id)
);

CREATE INDEX webhook_events_type_idx
  ON public.webhook_events (event_type, created_at DESC);

CREATE INDEX webhook_events_payload_subscription_idx
  ON public.webhook_events ((payload -> 'payment' ->> 'subscription'))
  WHERE event_type LIKE 'PAYMENT_%';

ALTER TABLE public.webhook_events ENABLE ROW LEVEL SECURITY;
-- No policies — table is only written by service_role from edge function.
-- The CONSTRAINT itself IS the security boundary.
-- Future admin/DPO reads happen via service_role; no authenticated/anon access.

-- Self-checks
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint
    WHERE conname = 'webhook_events_unique'
      AND conrelid = 'public.webhook_events'::regclass
  ) THEN
    RAISE EXCEPTION 'webhook_events UNIQUE constraint not created';
  END IF;

  IF NOT EXISTS (
    SELECT 1 FROM pg_indexes
    WHERE schemaname = 'public' AND tablename = 'webhook_events'
      AND indexname = 'webhook_events_payload_subscription_idx'
  ) THEN
    RAISE EXCEPTION 'webhook_events payload subscription index not created';
  END IF;
END $$;
```

**1.2 — Convention enforcement:**

- RLS enabled with NO policies = deny-all (service_role bypasses RLS by design)
- Inline `RAISE EXCEPTION` self-checks (Phase 1 convention)

-- Phase 2 W2a — HIGH-01 trial abuse defense + Asaas customer cache

-- 1. Add asaas_customer_id (denormalized cache; create-checkout-session writes it)
ALTER TABLE public.profiles
  ADD COLUMN IF NOT EXISTS asaas_customer_id TEXT;

CREATE INDEX IF NOT EXISTS profiles_asaas_customer_idx
  ON public.profiles (asaas_customer_id)
  WHERE asaas_customer_id IS NOT NULL;

-- 2. Add UNIQUE on cpf (HIGH-01 defense — Asaas dedupes too but DB is belt+suspenders)
-- Only enforce for non-null CPFs (cpf may be NULL for users that haven't filled the field)
DO $$
DECLARE
  duplicates BIGINT;
BEGIN
  -- Check for duplicates BEFORE adding constraint
  SELECT COUNT(*) INTO duplicates
  FROM (
    SELECT cpf FROM public.profiles WHERE cpf IS NOT NULL
    GROUP BY cpf HAVING COUNT(*) > 1
  ) d;

  IF duplicates > 0 THEN
    RAISE EXCEPTION 'profiles.cpf has % duplicate values — dedupe before applying UNIQUE constraint', duplicates;
  END IF;
END $$;

-- Use partial UNIQUE index (NULLs are allowed multiple times automatically in Postgres,
-- but partial index is more explicit about intent + lets us skip NULL rows in the index)
CREATE UNIQUE INDEX IF NOT EXISTS profiles_cpf_unique
  ON public.profiles (cpf)
  WHERE cpf IS NOT NULL;

-- 3. Self-checks
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM information_schema.columns
    WHERE table_schema = 'public' AND table_name = 'profiles' AND column_name = 'asaas_customer_id'
  ) THEN
    RAISE EXCEPTION 'asaas_customer_id not added';
  END IF;

  IF NOT EXISTS (
    SELECT 1 FROM pg_indexes
    WHERE schemaname = 'public' AND tablename = 'profiles' AND indexname = 'profiles_cpf_unique'
  ) THEN
    RAISE EXCEPTION 'profiles_cpf_unique not created';
  END IF;
END $$;
```

  if (a.length !== b.length) return false;

  for (let i = 0; i < a.length; i++) r |= a.charCodeAt(i) ^ b.charCodeAt(i);

  const eventType = body.event as string;

  const subscription = body.subscription;

async function handlePaymentSuccess(supabase: any, payment: any) {
  const subId = payment?.subscription;
  if (!subId) return;

  // Find subscription by Asaas ID OR by externalReference (our UUID)
  const { data: sub } = await supabase
    .from('user_subscriptions')
    .select('id, user_id, plan, status')
    .or(`asaas_subscription_id.eq.${subId},id.eq.${payment.externalReference ?? '00000000-0000-0000-0000-000000000000'}`)
    .maybeSingle();
  if (!sub) {
    console.warn('[asaas-webhook] PAYMENT_RECEIVED for unknown subscription', subId);
    return;
  }

  // Determine plan from externalReference UUID (created-checkout-session sets this)
  await supabase.from('user_subscriptions').update({
    is_active: true,
    status: 'active',
    last_paid_at: payment.paymentDate ?? new Date().toISOString(),

import { createClient } from 'https://esm.sh/@supabase/supabase-js@2';

const ASAAS_API_KEY = Deno.env.get('ASAAS_API_KEY');
const ASAAS_ENV = Deno.env.get('ASAAS_ENV') ?? 'sandbox';
const ASAAS_BASE = ASAAS_ENV === 'production'
  ? 'https://api.asaas.com/v3'
  : 'https://api-sandbox.asaas.com/v3';
const RECONCILE_AUTH_TOKEN = Deno.env.get('ASAAS_RECONCILE_AUTH_TOKEN');

function constantTimeEq(a: string, b: string): boolean {
  if (a.length !== b.length) return false;
  let r = 0;
  for (let i = 0; i < a.length; i++) r |= a.charCodeAt(i) ^ b.charCodeAt(i);
  return r === 0;
}

Deno.serve(async (req) => {
  // 1. Bearer auth (called by pg_cron with Vault-stored token)
  const auth = req.headers.get('Authorization') ?? '';
  const provided = auth.startsWith('Bearer ') ? auth.slice(7) : '';
  if (!RECONCILE_AUTH_TOKEN || !constantTimeEq(provided, RECONCILE_AUTH_TOKEN)) {
    return new Response('Forbidden', { status: 403 });
  }

  if (!ASAAS_API_KEY) {
    return new Response(JSON.stringify({ error: 'ASAAS_API_KEY unset' }), { status: 503 });
  }

  const supabase = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!,
  );

  // 2. Page all active Asaas subscriptions
  const asaasSubs: any[] = [];
  let offset = 0;
  while (true) {
    const res = await fetch(
      `${ASAAS_BASE}/subscriptions?status=ACTIVE&limit=100&offset=${offset}`,
      { headers: { 'access_token': ASAAS_API_KEY } },
    );
    if (!res.ok) {
      console.warn('[reconcile] Asaas error', res.status);
      break;
    }
    const page = await res.json();
    asaasSubs.push(...(page.data ?? []));
    if (!page.hasMore) break;
    offset += 100;
  }

  // 3. Compare against DB
  const { data: dbActive = [] } = await supabase
    .from('user_subscriptions')
    .select('id, user_id, asaas_subscription_id, status, plan')
    .eq('is_active', true);

  // Drift = rows where Asaas says ACTIVE but DB doesn't OR vice versa
  const asaasIds = new Set(asaasSubs.map((s) => s.id));
  const dbIds = new Set(dbActive.map((s: any) => s.asaas_subscription_id).filter(Boolean));

  const onlyInAsaas = asaasSubs.filter((s) => !dbIds.has(s.id));
  const onlyInDB = (dbActive as any[]).filter((s) => s.asaas_subscription_id && !asaasIds.has(s.asaas_subscription_id));

  const driftCount = onlyInAsaas.length + onlyInDB.length;
  const totalAsaas = asaasSubs.length;
  const driftPct = totalAsaas > 0 ? (driftCount / totalAsaas) * 100 : 0;

  // 4. Sentry alert if drift > 5%
  if (driftPct > 5) {
    console.warn(`[reconcile] DRIFT ALERT: ${driftCount}/${totalAsaas} rows out of sync (${driftPct.toFixed(1)}%)`);
    try {
      const Sentry: any = await import('https://esm.sh/@sentry/deno@7');
      Sentry.captureMessage(`Asaas reconcile drift: ${driftCount}/${totalAsaas} rows`, 'warning');
    } catch { /* Sentry deno SDK not installed yet; warn-only */ }
  }

  // 5. Self-heal: for `onlyInAsaas` (Asaas has it, DB doesn't) — synthesize a webhook_events row
  // for the latest payment so the webhook handler can replay it on next cron pass.
  // Simpler v1: just log; operator manually replays via curl.

  return new Response(JSON.stringify({
    asaas_count: totalAsaas,
    db_count: dbActive.length,
    drift_count: driftCount,
    drift_pct: Number(driftPct.toFixed(2)),
    only_in_asaas: onlyInAsaas.map((s) => s.id),
    only_in_db: onlyInDB.map((s) => s.asaas_subscription_id),
    asaas_env: ASAAS_ENV,
    generated_at: new Date().toISOString(),
  }), {
    headers: { 'Content-Type': 'application/json' },
  });
});
