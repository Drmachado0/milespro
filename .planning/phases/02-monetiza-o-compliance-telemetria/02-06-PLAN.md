```sql
-- Phase 2 W2b — TIER-01 (D-12) Free plan limits: 3 programas / 5 contas

-- 1. Drop and recreate user_programs INSERT policy with Free <3 limit
DO $$
BEGIN
  IF EXISTS (
    SELECT 1 FROM pg_policies
    WHERE schemaname = 'public' AND tablename = 'user_programs' AND policyname = 'user_programs_insert'
  ) THEN
    DROP POLICY user_programs_insert ON public.user_programs;
  END IF;
END $$;

CREATE POLICY user_programs_insert ON public.user_programs
  FOR INSERT
  WITH CHECK (
    auth.uid() = user_id
    AND (
      -- Pro+ unlimited
      public.has_plan(auth.uid(), 'pro'::public.subscription_plan)
      OR
      -- Free: enforce count < 3
      (SELECT COUNT(*) FROM public.user_programs WHERE user_id = auth.uid()) < 3
    )
  );

-- 2. Add a helpful error code via RAISE NOTICE in a wrapper function (optional but useful for client error parsing)
-- (RLS doesn't let us throw custom errors — clients see generic 42501 'permission denied'.
-- The error message is good enough for client to detect via i18n: "free_plan_limit_exceeded" appears in the policy name.)
-- Document the convention: clients should match on policy_name in the PG error structure for typed error handling.

-- 3. Repeat for program_accounts (if it exists per preflight)
-- [If program_accounts is NOT the right table name — adjust here]
DO $$
BEGIN
  IF EXISTS (SELECT 1 FROM information_schema.tables WHERE table_name = 'program_accounts') THEN
    -- Drop existing INSERT policy if any
    IF EXISTS (
      SELECT 1 FROM pg_policies WHERE tablename = 'program_accounts' AND policyname = 'program_accounts_insert'
    ) THEN
      DROP POLICY program_accounts_insert ON public.program_accounts;
    END IF;

    CREATE POLICY program_accounts_insert ON public.program_accounts
      FOR INSERT
      WITH CHECK (
        auth.uid() = user_id
        AND (
          public.has_plan(auth.uid(), 'pro'::public.subscription_plan)
          OR
          (SELECT COUNT(*) FROM public.program_accounts WHERE user_id = auth.uid()) < 5
        )
      );
  END IF;
END $$;

-- 4. Self-checks
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_policies WHERE tablename = 'user_programs' AND policyname = 'user_programs_insert'
  ) THEN
    RAISE EXCEPTION 'user_programs_insert policy not created';
  END IF;
END $$;
```

```ts
function constantTimeEq(a: string, b: string): boolean {
  if (a.length !== b.length) return false;
  let r = 0;
  for (let i = 0; i < a.length; i++) r |= a.charCodeAt(i) ^ b.charCodeAt(i);
  return r === 0;
}

Deno.serve(async (req) => {

  }
```
